cmake_minimum_required(VERSION 3.22)
project(workbench_c_reference VERSION 0.1.0 LANGUAGES C)
set(WORKBENCH_LANGUAGE_STANDARD 17 CACHE STRING "Target language standard")
set(CMAKE_C_STANDARD ${WORKBENCH_LANGUAGE_STANDARD})
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(WORKBENCH_COVERAGE "Instrument the canonical tests" OFF)
option(WORKBENCH_SANITIZERS "Check memory and undefined behavior" OFF)
option(WORKBENCH_PORTABLE_ONLY "Compile without optional SIMD" OFF)
set(WORKBENCH_LTO "off" CACHE STRING "Release link-time optimization: off, thin, or full")
set_property(CACHE WORKBENCH_LTO PROPERTY STRINGS off thin full)
if(NOT WORKBENCH_LTO MATCHES "^(off|thin|full)$")
  message(FATAL_ERROR "WORKBENCH_LTO must select off, thin, or full")
endif()
set(workbench_cpus "$ENV{WORKBENCH_CPUS}")
if(NOT workbench_cpus STREQUAL "" AND NOT workbench_cpus MATCHES "^[1-9][0-9]*$")
  message(FATAL_ERROR "WORKBENCH_CPUS must be a positive integer")
endif()
if(NOT WORKBENCH_LTO STREQUAL "off")
  if(WORKBENCH_COVERAGE OR WORKBENCH_SANITIZERS)
    message(FATAL_ERROR "Use a separate LTO build from coverage and sanitizers")
  endif()
  if(NOT CMAKE_BUILD_TYPE MATCHES "^(Release|RelWithDebInfo|MinSizeRel)$" OR CMAKE_CONFIGURATION_TYPES)
    message(FATAL_ERROR "LTO requires a separate optimized single-configuration build")
  endif()
  get_filename_component(compiler_bin "${CMAKE_C_COMPILER}" DIRECTORY)
  if(CMAKE_C_COMPILER_ID STREQUAL "Clang")
    find_program(lto_ar NAMES llvm-ar HINTS "${compiler_bin}" NO_CACHE REQUIRED)
    find_program(lto_ranlib NAMES llvm-ranlib HINTS "${compiler_bin}" NO_CACHE REQUIRED)
    if(WIN32)
      set(linker_name lld-link)
    elseif(APPLE)
      set(linker_name ld64.lld)
    else()
      set(linker_name ld.lld)
    endif()
    find_program(lto_linker NAMES ${linker_name} HINTS "${compiler_bin}" NO_CACHE REQUIRED)
    # Wrappers in managed environments may live separately; require matching LLVM releases.
    string(REPLACE "." "\\." compiler_version_pattern "${CMAKE_C_COMPILER_VERSION}")
    foreach(tool IN ITEMS "${lto_ar}" "${lto_ranlib}" "${lto_linker}")
      execute_process(COMMAND "${tool}" --version RESULT_VARIABLE code OUTPUT_VARIABLE version ERROR_VARIABLE diagnostic)
      if(NOT code EQUAL 0 OR NOT "${version}" MATCHES "(^|[ \n])${compiler_version_pattern}([ \n]|$)")
        message(FATAL_ERROR "LTO tool ${tool} must match Clang ${CMAKE_C_COMPILER_VERSION}: ${version}${diagnostic}")
      endif()
    endforeach()
    set(CMAKE_AR "${lto_ar}")
    set(CMAKE_RANLIB "${lto_ranlib}")
    set(CMAKE_C_COMPILER_AR "${lto_ar}")
    set(CMAKE_C_COMPILER_RANLIB "${lto_ranlib}")
    # Probe the exact requested mode; CMake's generic Clang IPO check chooses ThinLTO.
    include(CheckSourceCompiles)
    set(CMAKE_REQUIRED_FLAGS "-flto=${WORKBENCH_LTO}")
    set(CMAKE_REQUIRED_LINK_OPTIONS "-fuse-ld=${lto_linker}")
    unset(lto_supported CACHE)
    check_source_compiles(C "int main(void) { return 0; }" lto_supported)
    unset(CMAKE_REQUIRED_FLAGS)
    unset(CMAKE_REQUIRED_LINK_OPTIONS)
    if(NOT lto_supported)
      message(FATAL_ERROR "The selected toolchain cannot compile and link ${WORKBENCH_LTO} LTO")
    endif()
    add_compile_options(-flto=${WORKBENCH_LTO})
    add_link_options(-flto=${WORKBENCH_LTO} "-fuse-ld=${lto_linker}")
    if(NOT workbench_cpus STREQUAL "" AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
      add_link_options("-Wl,--threads=${workbench_cpus}" "-Wl,--lto-partitions=${workbench_cpus}")
      if(WORKBENCH_LTO STREQUAL "thin")
        add_link_options("-Wl,--thinlto-jobs=${workbench_cpus}")
      endif()
    endif()
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      # Explicit LLD bypasses managed linker wrappers. Retain only the SDK's linked shared runtimes.
      foreach(library IN LISTS CMAKE_C_IMPLICIT_LINK_LIBRARIES)
        unset(sdk_runtime)
        find_library(sdk_runtime NAMES "${library}" PATHS ${CMAKE_C_IMPLICIT_LINK_DIRECTORIES} NO_DEFAULT_PATH NO_CACHE)
        if(sdk_runtime MATCHES "\\.so(\\.|$)")
          file(REAL_PATH "${sdk_runtime}" sdk_runtime)
          get_filename_component(sdk_runtime_directory "${sdk_runtime}" DIRECTORY)
          list(APPEND CMAKE_BUILD_RPATH "${sdk_runtime_directory}")
        endif()
      endforeach()
      list(REMOVE_DUPLICATES CMAKE_BUILD_RPATH)
    endif()
  elseif(CMAKE_C_COMPILER_ID STREQUAL "GNU" AND WORKBENCH_LTO STREQUAL "full")
    find_program(lto_ar NAMES gcc-ar HINTS "${compiler_bin}" NO_DEFAULT_PATH NO_CACHE REQUIRED)
    find_program(lto_ranlib NAMES gcc-ranlib HINTS "${compiler_bin}" NO_DEFAULT_PATH NO_CACHE REQUIRED)
    set(CMAKE_C_COMPILER_AR "${lto_ar}")
    set(CMAKE_C_COMPILER_RANLIB "${lto_ranlib}")
    include(CheckIPOSupported)
    check_ipo_supported(LANGUAGES C)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
    if(NOT workbench_cpus STREQUAL "")
      add_link_options("-flto=${workbench_cpus}")
    endif()
  else()
    message(FATAL_ERROR "The selected compiler does not support the requested LTO mode")
  endif()
endif()
if(NOT CMAKE_C_COMPILER_ID MATCHES "^(Clang|GNU)$")
  message(FATAL_ERROR "Select a declared Clang or GCC toolchain")
endif()
if((WORKBENCH_COMPILER_FAMILY STREQUAL "clang" AND NOT CMAKE_C_COMPILER_ID STREQUAL "Clang") OR
   (WORKBENCH_COMPILER_FAMILY STREQUAL "gcc" AND NOT CMAKE_C_COMPILER_ID STREQUAL "GNU"))
  message(FATAL_ERROR "Compiler does not match the selected family")
endif()
if(WORKBENCH_COMPILER_MINIMUM AND CMAKE_C_COMPILER_VERSION VERSION_LESS WORKBENCH_COMPILER_MINIMUM)
  message(FATAL_ERROR "Compiler is below the configured minimum")
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Wformat=2 -Wundef -Werror)
if(WORKBENCH_COVERAGE)
  if(NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "LLVM coverage requires Clang")
  endif()
  add_compile_options(-fprofile-instr-generate -fcoverage-mapping)
  add_link_options(-fprofile-instr-generate)
endif()
if(WORKBENCH_SANITIZERS)
  if(WIN32)
    add_compile_options(-fsanitize=address)
    add_link_options(-fsanitize=address)
  else()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
add_library(byte_count STATIC src/count.c)
target_include_directories(byte_count PUBLIC include)
if(WIN32 AND NOT WORKBENCH_PORTABLE_ONLY)
  # The MSVC target does not automatically link Clang's CPU-dispatch builtins.
  execute_process(COMMAND "${CMAKE_C_COMPILER}" --rtlib=compiler-rt --print-libgcc-file-name
    RESULT_VARIABLE builtins_status OUTPUT_VARIABLE builtins OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT builtins_status EQUAL 0 OR NOT EXISTS "${builtins}")
    message(FATAL_ERROR "The selected Clang installation must provide its compiler-rt builtins")
  endif()
  target_link_libraries(byte_count PUBLIC "${builtins}")
endif()
if(WORKBENCH_PORTABLE_ONLY)
  target_compile_definitions(byte_count PRIVATE WORKBENCH_PORTABLE_ONLY)
endif()
find_package(yyjson CONFIG REQUIRED)
add_library(count_json STATIC src/count_json.c)
target_include_directories(count_json PUBLIC include)
target_link_libraries(count_json PUBLIC yyjson::yyjson)
add_executable(count src/main.c)
target_link_libraries(count PRIVATE byte_count count_json)
add_executable(count_test tests/count_test.c)
target_link_libraries(count_test PRIVATE byte_count)
add_executable(count_json_test tests/count_json_test.c)
target_link_libraries(count_json_test PRIVATE count_json)
enable_testing()
add_test(NAME json_failures COMMAND count_json_test ${CMAKE_CURRENT_SOURCE_DIR}/README.md)
foreach(backend scalar auto avx2)
  add_test(NAME count_${backend} COMMAND count_test ${backend})
endforeach()
set_tests_properties(count_avx2 PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME cli COMMAND ${CMAKE_COMMAND} -DPROGRAM=$<TARGET_FILE:count> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
