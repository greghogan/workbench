cmake_minimum_required(VERSION 3.22)
project(workbench_cpp_reference VERSION 0.1.0 LANGUAGES CXX)
set(WORKBENCH_LANGUAGE_STANDARD 20 CACHE STRING "Target language standard")
set(CMAKE_CXX_STANDARD ${WORKBENCH_LANGUAGE_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(WORKBENCH_COVERAGE "Instrument the canonical tests" OFF)
option(WORKBENCH_SANITIZERS "Check memory and undefined behavior" OFF)
option(WORKBENCH_PORTABLE_ONLY "Compile without optional SIMD" OFF)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "^(Clang|GNU)$")
  message(FATAL_ERROR "Select a declared Clang or GCC toolchain")
endif()
if((WORKBENCH_COMPILER_FAMILY STREQUAL "clang" AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang") OR
   (WORKBENCH_COMPILER_FAMILY STREQUAL "gcc" AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU"))
  message(FATAL_ERROR "Compiler does not match the selected family")
endif()
if(WORKBENCH_COMPILER_MINIMUM AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS WORKBENCH_COMPILER_MINIMUM)
  message(FATAL_ERROR "Compiler is below the configured minimum")
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Wformat=2 -Wundef -Werror)
if(WORKBENCH_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "LLVM coverage requires Clang")
  endif()
  add_compile_options(-fprofile-instr-generate -fcoverage-mapping)
  add_link_options(-fprofile-instr-generate)
endif()
if(WORKBENCH_SANITIZERS)
  if(WIN32)
    add_compile_options(-fsanitize=address)
    add_link_options(-fsanitize=address)
  else()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
add_library(byte_count src/count.cpp)
target_include_directories(byte_count PUBLIC include)
if(WORKBENCH_PORTABLE_ONLY)
  target_compile_definitions(byte_count PRIVATE WORKBENCH_PORTABLE_ONLY)
endif()
find_package(fmt CONFIG REQUIRED)
add_executable(count src/main.cpp)
target_link_libraries(count PRIVATE byte_count fmt::fmt)
add_executable(count_test tests/count_test.cpp)
target_link_libraries(count_test PRIVATE byte_count)
enable_testing()
foreach(backend scalar auto avx2)
  add_test(NAME count_${backend} COMMAND count_test ${backend})
endforeach()
set_tests_properties(count_avx2 PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME cli COMMAND ${CMAKE_COMMAND} -DPROGRAM=$<TARGET_FILE:count> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
