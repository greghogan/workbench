cmake_minimum_required(VERSION 3.22)
project(workbench_cpp_reference VERSION 0.1.0 LANGUAGES CXX)
set(WORKBENCH_LANGUAGE_STANDARD 20 CACHE STRING "Target language standard")
set(CMAKE_CXX_STANDARD ${WORKBENCH_LANGUAGE_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# This reference uses headers, not C++ modules; probes must not require clang-scan-deps.
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
include(cmake/WorkbenchCompiler.cmake)
workbench_compiler_checks(CXX)
add_library(byte_count STATIC src/count.cpp src/count_mvsc.cpp)
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" AND NOT WORKBENCH_PORTABLE_ONLY)
  set_source_files_properties(src/count_mvsc.cpp PROPERTIES COMPILE_OPTIONS /arch:AVX2)
endif()
target_include_directories(byte_count PUBLIC include)
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND NOT WORKBENCH_PORTABLE_ONLY)
  # The MSVC target does not automatically link Clang's CPU-dispatch builtins.
  execute_process(COMMAND "${CMAKE_CXX_COMPILER}" --rtlib=compiler-rt --print-libgcc-file-name
    RESULT_VARIABLE builtins_status OUTPUT_VARIABLE builtins OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT builtins_status EQUAL 0 OR NOT EXISTS "${builtins}")
    message(FATAL_ERROR "The selected Clang installation must provide its compiler-rt builtins")
  endif()
  target_link_libraries(byte_count PUBLIC "${builtins}")
endif()
if(WORKBENCH_PORTABLE_ONLY)
  target_compile_definitions(byte_count PRIVATE WORKBENCH_PORTABLE_ONLY)
endif()
find_package(fmt CONFIG REQUIRED)
add_executable(count src/main.cpp)
target_link_libraries(count PRIVATE byte_count fmt::fmt)
add_executable(count_test tests/count_test.cpp)
target_link_libraries(count_test PRIVATE byte_count)
enable_testing()
foreach(backend scalar auto avx2)
  add_test(NAME count_${backend} COMMAND count_test ${backend})
endforeach()
set_tests_properties(count_avx2 PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME cli COMMAND ${CMAKE_COMMAND} -DPROGRAM=$<TARGET_FILE:count> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)

include(GNUInstallDirs)
install(TARGETS count RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/licenses/${PROJECT_NAME})

add_executable(cpu_test tests/cpu_test.c)
target_include_directories(cpu_test PRIVATE include)
set_source_files_properties(tests/cpu_test.c PROPERTIES LANGUAGE CXX)
add_test(NAME cpu_capabilities COMMAND cpu_test)
